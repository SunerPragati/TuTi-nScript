-- Hàm gọi Lôi Kiếp
function GoiLoiKiep(soLuong)
	local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	for i = 1, soLuong do
		local bolt = Instance.new("Part", workspace)
		bolt.Anchored = true
		bolt.Size = Vector3.new(1, 30, 1)
		bolt.BrickColor = BrickColor.new("Electric blue")
		bolt.Material = Enum.Material.Neon
		bolt.CFrame = hrp.CFrame * CFrame.new(math.random(-3,3), 20, math.random(-3,3))
		game:GetService("Debris"):AddItem(bolt, 0.5)
		wait(0.3)
	end
end

-- Cooldown kỹ năng
local cooldowns = {}
function coTheDungKyNang(ten, delay)
	if cooldowns[ten] and tick() - cooldowns[ten] < delay then return false end
	cooldowns[ten] = tick()
	return true
end

-- 🗡️ NGỰ KIẾM PHI HÀNH – PHẦN 4.2: CHẾ ĐỘ BAY VÀ ĐIỀU KHIỂN (CẢI TIẾN)
local function bayTrenKiem()
	if TuTienData.CanhGioi == "Luyện Khí" then
		warn("Luyện Khí chưa thể Ngự Kiếm!")
		return
	end
	local kiem = taoKiemBay()
	local weld = Instance.new("WeldConstraint")
	weld.Part0 = kiem
	weld.Part1 = char.HumanoidRootPart
	weld.Parent = kiem

	local bodyVelocity = Instance.new("BodyVelocity")
	bodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
	bodyVelocity.Velocity = Vector3.new(0,0,0)
	bodyVelocity.Parent = char.HumanoidRootPart

	local speed = 60
	local uis = game:GetService("UserInputService")
	local flying = true

	uis.InputBegan:Connect(function(input, gpe)
		if gpe then return end
		if input.KeyCode == Enum.KeyCode.W then
			bodyVelocity.Velocity = char.HumanoidRootPart.CFrame.LookVector * speed
		elseif input.KeyCode == Enum.KeyCode.S then
			bodyVelocity.Velocity = -char.HumanoidRootPart.CFrame.LookVector * speed
		elseif input.KeyCode == Enum.KeyCode.Space then
			bodyVelocity.Velocity = Vector3.new(0, speed, 0)
		elseif input.KeyCode == Enum.KeyCode.LeftControl then
			bodyVelocity.Velocity = Vector3.new(0, -speed, 0)
		end
	end)

	uis.InputEnded:Connect(function(input, gpe)
		if gpe then return end
		bodyVelocity.Velocity = Vector3.new(0,0,0)
	end)
end

	hum.Health = hum.MaxHealth
	hum.WalkSpeed = 10 + cap
	hum.DisplayName = "👹 Ma Tu"

	-- Tự động tìm người chơi gần nhất
	while hum.Health > 0 do
		wait(1)
		local players = game.Players:GetPlayers()
		local nearest, dist = nil, 999
		for _, plr in pairs(players) do
			if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
				local d = (plr.Character.HumanoidRootPart.Position - part.Position).Magnitude
				if d < dist then
					dist = d
					nearest = plr
				end
			end
		end

		if nearest and dist < 100 then
			local hrp = nearest.Character.HumanoidRootPart
			local dir = (hrp.Position - part.Position).Unit
			part.CFrame = CFrame.new(part.Position + dir * 2)
		end
	end
end

local function nangCap(vatPham)
	local cap = vatPham.CapDo or 1
	local ngLieu = "Tinh Thiết"
	local soLuong = cap * 2

	if not CoDayDuNguyenLieu(ngLieu, soLuong) then
		warn("Thiếu nguyên liệu nâng cấp!")
		return
	end

	local tyLe = math.max(60 - (cap * 5), 10) -- Tỷ lệ giảm theo cấp
	local ketQua = math.random(1, 100)

	if ketQua <= tyLe then
		vatPham.CapDo = cap + 1
		DatThanhTuu("Tăng cấp "..vatPham.Ten.." lên +"..vatPham.CapDo)
		warn("Nâng cấp thành công!")
	else
		if math.random() < 0.3 then
			vatPham.CapDo = math.max(1, cap - 1)
			warn("Thất bại! Bị giảm cấp.")
		else
			warn("Nâng cấp thất bại. Mất nguyên liệu.")
		end
	end
end

local function tinhSatThuong(vatPham)
	local base = 20
	local cap = vatPham.CapDo or 1
	local phamChat = vatPham.PhamChat or "Thường"

	local bonus = 0
	if phamChat == "Tốt" then bonus = 1.2 end
	if phamChat == "Hiếm" then bonus = 1.5 end
	if phamChat == "Huyền Thoại" then bonus = 2 end
	if phamChat == "Thần Thoại" then bonus = 3 end

	return math.floor(base * cap * bonus)
end

-- Boss dùng Đạo – và Gầm Triệu Thiên Kiếp
local function SpawnDaoBoss(daoType)
	local boss = Instance.new("Model", workspace)
	boss.Name = "DaoBoss"
	local head = Instance.new("Part", boss)
	head.Name = "Head"
	head.Size = Vector3.new(4, 4, 4)
	head.Position = Vector3.new(0, 20, 0)
	head.Anchored = false
	local humanoid = Instance.new("Humanoid", boss)
	boss:SetPrimaryPartC(head)
	boss.PrimaryPart = head
	
	humanoid.Died:Connect(function()
		player:SetAttribute("DaoBossDefeated", daoType)
	end)

	spawn(function()
		while humanoid.Health > 0 do
			wait(10)
			local viTri = head.Position
			HieuUngDaiDao({GetAttribute=function() return daoType end}, viTri)

			-- GẦM gọi Thiên Kiếp
			local sound = Instance.new("Sound", head)
			sound.SoundId = "rbxassetid://12222216" -- Gầm trời
			sound:Play()

			local thunder = Instance.new("Explosion", workspace)
			thunder.Position = viTri + Vector3.new(0, 10, 0)
			thunder.BlastRadius = 5
		end
	end)
end

-- Boss Bay Biết Thi Triển Đạo
local function SpawnDaoBoss(daoType)
	local boss = Instance.new("Model", workspace)
	boss.Name = "DaoBoss"
	local root = Instance.new("Part", boss)
	root.Name = "HumanoidRootPart"
	root.Size = Vector3.new(4, 1, 4)
	root.Position = Vector3.new(0, 50, 0)
	root.Anchored = false
	local head = Instance.new("Part", boss)
	head.Name = "Head"
	head.Size = Vector3.new(4, 4, 4)
	head.Position = Vector3.new(0, 54, 0)
	head.Anchored = false
	local humanoid = Instance.new("Humanoid", boss)
	boss:SetPrimaryPartC(root)

	-- Bay theo sin
	spawn(function()
		local t = 0
		while humanoid.Health > 0 do
			t += 0.1
			root.CFrame = CFrame.new(0, 50 + math.sin(t)*10, 0)
			wait(0.1)
		end
	end)

	-- Gầm + Thi triển kỹ năng
	spawn(function()
		while humanoid.Health > 0 do
			wait(10)
			local viTri = root.Position
			HieuUngDaiDao({GetAttribute=function() return daoType end}, viTri)
			local sfx = Instance.new("Sound", head)
			sfx.SoundId = "rbxassetid://12222216"
			sfx:Play()
		end
	end)
end
